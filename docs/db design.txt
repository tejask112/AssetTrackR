table: portfolio
- field:user_id, type:varchar(128), not null
- field:stock_id, type:integer, not null
- field:total_quantity, type:numeric, not null, (note: total number of shares user currently owns, including reserved shares for pending sell, available quantity = total_quantity - reserved_quantity)
- field:reserved_quantity, type:numeric, not null, (note: nuber of shares within total_quantity that are committed/locked to a pending sell order.)
- primary key on (user_id, stock_id)
- foreign key on (user_id) to 'users'
- foreign key on (stock_id) to 'stocks'

table: portfolio_history
- field:user_id, type:varchar(128), not null
- field:date_recorded, type:timestamptz, not null
- field:cash_value, type:numeric, not null
- field:portfolio_value, type:numeric, not null
- primary key on (user_id, date_recorded)
- foreign key on (user_id) to 'users'
- index on (user_id, date_recorded desc)
- note: total account value is cash_value + portfolio_value

table: orders
- field:order_id, type:uuid, not null
- field:user_id, type:varchar(128), not null
- field:stock_id, type:integer, not null
- field:side, type:text, not null, (note: either 'buy' or 'sell')
- field:order_type, type:text, not null, (note: either 'market', 'limit', 'stop')
- field:status, type:text, not null, (note: either 'executed', 'queued', 'pending', 'cancelled' or 'rejected')
- field:quantity, type:numeric, not null
- field:reserved_cash, type:numeric
- field:trigger_price, type:numeric, (note: the price that activates stop loss/take profit, null for normal market orders)
- field:date_submitted, type:timestamptz, not null
- field:date_executed, type:timestamptz
- field:execution_price, type:numeric, (note: individual price per stock, not total execution price)
- field:rejection_reason, type:text
- field:idempotency_key, type:uuid, not null
- primary key on (order_id)
- foreign key on (user_id) to 'users'
- foreign key on (stock_id) to 'stocks'
- unique constraint (user_id, idempotency_key)
- index on (user_id, date_submitted DESC)
- index on (user_id, stock_id, date_submitted DESC)
- index on (stock_id) where status = 'queued'
- index on (stock_id) where status = 'pending'

note on orders.status:
executed = the entire order was successfully executed.
queued = an accepted market order waiting specifically only for the next market open.
pending = an accepted limit or stop order waiting for its price trigger.
cancelled = a queued or pending order cancelled by the user.
rejected = the system wasn't accepted by the user due to some condition.

table: ai_portfolio_summaries
- field:summary_id, type:uuid, not null
- field:user_id, type:varchar(128), not null
- field:status, type:text, not null, (note: either 'generating', 'completed' or 'failed')
- field:idempotency_key, type:uuid, not null
- field:model, type:text, not null
- field:portfolio_snapshot, type:jsonb, not null
- field:summary, type:text
- field:error_message, type:text
- field:requested_at, type:timestamptz, not null
- primary key on (summary_id)
- foreign key on (user_id) to 'users'
- index on (user_id, requested_at desc)
- unique constraint on (user_id, idempotency_key)
- unique constraint (index) on (user_id, status) where status='generating' -> don't want user to request twice
    - lazy timeout logic if status='generating' and worker unexpectedly shuts down
    1. leave the status as 'generating'
    2. next time user clicks 'generate ai summary' button, check the existing row's 'requested_at'
    3. if it is less than 3 min old, treat it as active and dont create new summary.
    4. if it is older than 3 mins, assume the generation was interrupted
    5. update abandoned row to status='failed' and add appropriate 'error_message'
    6. insert new row with status='generating' and make new ai request
    7. store it once done and mark as status='completed'
